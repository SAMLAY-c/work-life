# 浏览器自动化复制行的东西
```
(async function() {
    // === 配置项 ===
    const WAIT_TIME = 5000; // 等待时间，单位毫秒（5000 = 5秒）
    const FILENAME = 'chat_data.csv';

    // === 存储数据的变量 ===
    // 添加 CSV 表头，\uFEFF 是为了让 Excel 正确识别中文编码
    let csvContent = '\uFEFF"用户提问","ai回答"\n'; 
    let pageCount = 1;

    // === 辅助函数：等待 ===
    const sleep = (ms) => new Promise(r => setTimeout(r, ms));

    // === 辅助函数：处理 CSV 特殊字符 (英文逗号, 换行符, 双引号) ===
    const escapeCsv = (text) => {
        if (!text) return "";
        let cleanText = text.replace(/"/g, '""'); // 将双引号替换为两个双引号
        return `"${cleanText}"`;
    };

    console.log("🚀 开始自动化抓取...");

    while (true) {
        console.log(`正在处理第 ${pageCount} 页...`);

        // 1. 获取当前页面的所有行
        const rows = document.querySelectorAll('tr.ant-table-row');
        
        if (rows.length === 0) {
            console.warn("未找到数据行，可能页面未加载或已结束。");
            break;
        }

        let rowDataCount = 0;

        // 2. 遍历每一行提取数据
        rows.forEach(row => {
            const cells = row.querySelectorAll('td');
            if (cells.length < 4) return; // 防止空行

            // --- 提取用户提问 (第3列, 也就是索引2) ---
            // 根据你的HTML，提问在第3列，且在 "接收文本消息" 后面
            // 我们查找该单元格内具有 line-clamp 样式的 div
            const questionCell = cells[2];
            // 尝试查找包含具体样式的div，或者是该列中第二个主要的文本块
            let questionDiv = questionCell.querySelector('div[style*="-webkit-line-clamp: 2"]');
            
            // 如果没找到特定样式的，尝试找该列最后一个 div (通常是内容)
            if (!questionDiv) {
                 const divs = questionCell.querySelectorAll('.line-clamp-1');
                 if(divs.length > 0) questionDiv = divs[divs.length - 1];
            }
            
            const questionText = questionDiv ? questionDiv.innerText : "未提取到提问";

            // --- 提取AI回答 (第4列, 也就是索引3) ---
            const answerCell = cells[3];
            // 回答在 "发送消息" 后面
            let answerDiv = answerCell.querySelector('div[style*="-webkit-line-clamp: 1"]');
            // 如果上面的选择器没选中，尝试通过 flex 结构找
            if (!answerDiv) {
                 const divs = answerCell.querySelectorAll('.line-clamp-1');
                 if(divs.length > 0) answerDiv = divs[divs.length - 1];
            }

            const answerText = answerDiv ? answerDiv.innerText : "未提取到回答";

            // 添加到 CSV 字符串
            csvContent += `${escapeCsv(questionText)},${escapeCsv(answerText)}\n`;
            rowDataCount++;
        });

        console.log(`✅ 第 ${pageCount} 页提取了 ${rowDataCount} 条数据。`);

        // 3. 寻找并点击“下一页”按钮
        // Ant Design 的下一页按钮通常在 li.ant-pagination-next 中
        const nextButtonLi = document.querySelector('li.ant-pagination-next');
        const nextButton = document.querySelector('li.ant-pagination-next .ant-pagination-item-link');

        // 检查是否禁用（没有下一页了）
        // ant-pagination-disabled 是 Ant Design 禁用按钮的类名
        // aria-disabled 是辅助功能属性
        if (!nextButtonLi || nextButtonLi.classList.contains('ant-pagination-disabled') || nextButtonLi.getAttribute('aria-disabled') === 'true') {
            console.log("🏁 已到达最后一页，停止抓取。");
            break;
        }

        if (nextButton) {
            nextButton.click();
            pageCount++;
            
            console.log(`⏳ 等待 ${WAIT_TIME/1000} 秒加载下一页...`);
            await sleep(WAIT_TIME);
        } else {
            console.error("未找到下一页按钮，脚本停止。");
            break;
        }
    }

    // 4. 下载 CSV 文件
    console.log("💾 正在导出 CSV 文件...");
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement("a");
    const url = URL.createObjectURL(blob);
    link.setAttribute("href", url);
    link.setAttribute("download", FILENAME);
    link.style.visibility = 'hidden';
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);

    console.log("🎉 完成！文件应已开始下载。");

})();
```
