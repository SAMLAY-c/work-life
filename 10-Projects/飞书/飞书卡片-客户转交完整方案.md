## 第一步：飞书应用配置

### 1.1 创建应用并开启权限

| 步骤 | 操作 | 位置 |
|------|------|------|
| 1 | 创建企业自建应用 | [开发者后台](https://open.feishu.cn/) |
| 2 | 开启机器人能力 | 应用详情 → 机器人 |
| 3 | 获取 App ID 和 App Secret | 应用详情 → 凭证与基础信息 |
| 4 | 订阅事件 `card.action.trigger` | 应用详情 → 事件订阅 |
| 5 | 添加权限（见下表） | 应用详情 → 权限管理 |
| 6 | 发布应用 | 应用详情 → 版本管理与发布 |

### 1.2 需要的权限

搜索并添加：
- `im:chat:readonly` - 获取群聊信息
- `im:chat.member_id:readonly` - 获取群成员
- `im:message:send` - 发送消息
- `bitable:record:write` - 更新多维表格

## 第二步：后端代码（纯 Node.js，零依赖）

创建文件 `feishu-server.js`：

```javascript
const http = require('http');
const https = require('https');

// ==================== 配置 ====================
const CONFIG = {
  PORT: 5000,
  APP_ID: 'cli_你的AppID',           // 替换为你的 App ID
  APP_SECRET: '你的AppSecret',       // 替换为你的 App Secret
  CHAT_ID: 'oc_你的群聊ID',          // 目标群聊 ID
  BASE_ID: 'bascn你的BaseID',       // 多维表格 ID
  TABLE_ID: 'tbl你的TableID'        // 表格 ID
};

// Token 缓存
let tokenCache = { token: null, expireTime: 0 };

// ==================== 工具函数 ====================

// 发送 HTTPS 请求
function request(url, options = {}, data = null) {
  return new Promise((resolve, reject) => {
    const client = url.startsWith('https') ? https : http;
    const req = client.request(url, options, (res) => {
      let body = '';
      res.on('data', chunk => body += chunk);
      res.on('end', () => {
        try { resolve(JSON.parse(body)); } catch { resolve(body); }
      });
    });
    req.on('error', reject);
    if (data) req.write(data);
    req.end();
  });
}

// 获取 tenant_access_token
async function getTenantToken() {
  const now = Date.now();
  if (tokenCache.token && now < tokenCache.expireTime) return tokenCache.token;

  const result = await request(
    'https://open.feishu.cn/open-apis/auth/v3/tenant_access_token/internal/',
    { method: 'POST', headers: { 'Content-Type': 'application/json' } },
    JSON.stringify({ app_id: CONFIG.APP_ID, app_secret: CONFIG.APP_SECRET })
  );

  if (result.tenant_access_token) {
    tokenCache.token = result.tenant_access_token;
    tokenCache.expireTime = now + (result.expire - 300) * 1000;
  }
  return tokenCache.token;
}

// 获取群成员列表
async function getChatMembers(chatId) {
  const token = await getTenantToken();
  const url = `https://open.feishu.cn/open-apis/im/v1/chats/${chatId}/members`;
  return request(url, { headers: { 'Authorization': `Bearer ${token}` } });
}

// 更新多维表格
async function updateBitable(recordId, fields) {
  const token = await getTenantToken();
  const url = `https://open.feishu.cn/open-apis/bitable/v1/apps/${CONFIG.BASE_ID}/tables/${CONFIG.TABLE_ID}/records/${recordId}`;
  return request(url, {
    method: 'PUT',
    headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }
  }, JSON.stringify({ fields }));
}

// 发送卡片消息
async function sendCardMessage(chatId, card) {
  const token = await getTenantToken();
  const url = 'https://open.feishu.cn/open-apis/im/v1/messages?receive_id_type=chat_id';
  return request(url, {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }
  }, JSON.stringify({
    receive_id: chatId,
    msg_type: 'interactive',
    content: JSON.stringify(card)
  }));
}

// ==================== 构建卡片 ====================

function buildCustomerCard(customer, assignedUser) {
  const now = new Date();
  const dateStr = `${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}-${String(now.getDate()).padStart(2,'0')}`;
  
  return {
    schema: '2.0',
    config: {
      streaming_mode: false,
      summary: { content: `🎯 SDR转交通知：${customer.name}` },
      enable_forward: true,
      update_multi: true,
      width_mode: 'fill'
    },
    header: {
      template: 'blue',
      title: { tag: 'plain_text', content: '🎯 SDR 客户转交通知' },
      subtitle: { tag: 'plain_text', content: `来自 SDR 机器人 · ${dateStr}` }
    },
    body: {
      direction: 'vertical',
      elements: [
        { tag: 'markdown', content: '**👤 客户信息**' },
        {
          tag: 'column_set',
          columns: [
            {
              tag: 'column',
              width: 'weighted',
              weight: 1,
              elements: [
                { tag: 'markdown', content: `**客户姓名**\n<font color="grey">${customer.name}</font>` }
              ]
            },
            {
              tag: 'column',
              width: 'weighted',
              weight: 1,
              elements: [
                { tag: 'markdown', content: `**联系电话**\n<font color="grey">${customer.phone}</font>` }
              ]
            }
          ]
        },
        { tag: 'hr' },
        {
          tag: 'markdown',
          content: `**📋 需求详情**\n\n> 🏷️ 产品需求：${customer.product}\n> 💰 预算范围：${customer.budget}\n> ⏰ 跟进优先级：${customer.priority}`
        },
        { tag: 'hr' },
        { tag: 'markdown', content: `**💡 SDR 备注**\n\n${customer.remark}` },
        { tag: 'hr' },
        {
          tag: 'markdown',
          content: `**👇 分配销售**\n\n<at id="${assignedUser.id}">${assignedUser.name}</at> 这是你的新客户，请尽快跟进！`
        },
        { tag: 'hr' },
        {
          tag: 'column_set',
          columns: [
            {
              tag: 'column',
              width: 'weighted',
              weight: 1,
              elements: [
                {
                  tag: 'button',
                  text: { tag: 'plain_text', content: '📞 一键拨打' },
                  type: 'primary',
                  behaviors: [{ type: 'open_url', default_url: `tel:${customer.phoneRaw}` }]
                }
              ]
            },
            {
              tag: 'column',
              width: 'weighted',
              weight: 1,
              elements: [
                {
                  tag: 'button',

                  type: 'default',
                  behaviors: [{
                    type: 'callback',
                    value: {
                      action: 'show_status_options',
                      record_id: customer.recordId,
                      assigned_user_id: assignedUser.id,
                      assigned_user_name: assignedUser.name
                    }
                  }]
                }
              ]
            }
          ]
        }
      ]
    }
  };
}

// ==================== HTTP 服务 ====================

const server = http.createServer(async (req, res) => {
  // CORS
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');
  if (req.method === 'OPTIONS') { res.writeHead(200); res.end(); return; }

  let body = '';
  req.on('data', chunk => body += chunk);
  req.on('end', async () => {
    let jsonBody = {};
    try { jsonBody = JSON.parse(body); } catch {}

    try {
      // 获取群成员
      if (req.url === '/api/members' && req.method === 'GET') {
        const result = await getChatMembers(CONFIG.CHAT_ID);
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify(result));
        return;
      }

      // 发送客户转交卡片
      if (req.url === '/api/send-card' && req.method === 'POST') {
        const { customer, assignedUser } = jsonBody;
        const card = buildCustomerCard(customer, assignedUser);
        const result = await sendCardMessage(CONFIG.CHAT_ID, card);
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify(result));
        return;
      }

      // 卡片回调处理
      if (req.url === '/webhook/feishu-card' && req.method === 'POST') {
        await handleCardCallback(jsonBody, res);
        return;
      }

      // 健康检查
      if (req.url === '/' && req.method === 'GET') {
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ status: 'ok', time: new Date().toISOString() }));
        return;
      }

      res.writeHead(404); res.end('Not Found');
    } catch (err) {
      console.error('[错误]', err);
      res.writeHead(500, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ error: err.message }));
    }
  });
});

// 处理卡片回调
async function handleCardCallback(data, res) {
  const event = data.event || {};
  const operator = event.operator || {};
  const clickerName = operator.name || '未知用户';
  
  const value = (event.action && event.action.value) || {};
  const action = value.action;
  const recordId = value.record_id;

  console.log(`[点击] ${clickerName} 点击了 ${action}`);

  // 第一步：显示状态选择
  if (action === 'show_status_options') {
    const response = {
      toast: { type: 'info', content: '请选择新状态' },
      card: {
        schema: '2.0',
        header: {
          template: 'blue',
          title: { tag: 'plain_text', content: '📝 选择新状态' }
        },
        body: {
          direction: 'vertical',
          elements: [
            { tag: 'markdown', content: `**操作人：**${clickerName}` },
            { tag: 'hr' },
            {
              tag: 'column_set',
              columns: [
                {
                  tag: 'column',
                  width: 'weighted',
                  weight: 1,
                  elements: [{
                    tag: 'button',
                    text: { tag: 'plain_text', content: '📞 跟进中' },
                    type: 'primary',
                    behaviors: [{
                      type: 'callback',
                      value: { action: 'set_status', record_id: recordId, status: '跟进中', operator: clickerName }
                    }]
                  }]
                },
                {
                  tag: 'column',
                  width: 'weighted',
                  weight: 1,
                  elements: [{
                    tag: 'button',
                    text: { tag: 'plain_text', content: '✅ 已成交' },
                    type: 'primary',
                    behaviors: [{
                      type: 'callback',
                      value: { action: 'set_status', record_id: recordId, status: '已成交', operator: clickerName }
                    }]
                  }]
                },
                {
                  tag: 'column',
                  width: 'weighted',
                  weight: 1,
                  elements: [{
                    tag: 'button',
                    text: { tag: 'plain_text', content: '❌ 已放弃' },
                    type: 'danger',
                    behaviors: [{
                      type: 'callback',
                      value: { action: 'set_status', record_id: recordId, status: '已放弃', operator: clickerName }
                    }]
                  }]
                }
              ]
            }
          ]
        }
      }
    };
    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify(response));
    return;
  }

  // 第二步：设置状态并更新表格
  if (action === 'set_status') {
    const status = value.status;
    const operator = value.operator;
    const now = new Date();
    const timeStr = `${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}-${String(now.getDate()).padStart(2,'0')}T${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}:${String(now.getSeconds()).padStart(2,'0')}`;

    // 更新多维表格
    let updateResult = { code: 0 };
    if (CONFIG.BASE_ID && CONFIG.TABLE_ID) {
      updateResult = await updateBitable(recordId, {
        '当前状态': status,
        '最后更新时间': timeStr,
        '确认人': operator
      });
    }
    console.log('[更新表格]', status, updateResult);

    const response = {
      toast: { type: 'success', content: `✅ 状态已更新为：${status}` },
      card: {
        schema: '2.0',
        header: {
          template: 'green',
          title: { tag: 'plain_text', content: '✅ 状态更新成功' }
        },
        body: {
          direction: 'vertical',
          elements: [
            { tag: 'markdown', content: `**操作人：**${operator}\n**新状态：**${status}\n**时间：**${timeStr}` }
          ]
        }
      }
    };
    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify(response));
    return;
  }

  res.writeHead(200, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify({ toast: { type: 'info', content: '已收到' } }));
}

// 启动
server.listen(CONFIG.PORT, () => {
  console.log('========================================');
  console.log('🚀 SDR 客户转交服务已启动');
  console.log(`📍 地址: http://localhost:${CONFIG.PORT}`);
  console.log('');
  console.log('📋 API 接口:');
  console.log(`   GET  /api/members       - 获取群成员列表`);
  console.log(`   POST /api/send-card     - 发送客户转交卡片`);
  console.log(`   POST /webhook/feishu-card - 飞书卡片回调`);
  console.log('');
  console.log('⚠️  请修改 CONFIG 中的配置项');
  console.log('========================================');
});
```

## 第三步：多维表格准备

### 3.1 创建表格

**表格名称**：SDR 客户转交记录

| 字段名 | 字段类型 | 说明 |
|--------|---------|------|
| 客户姓名 | 文本 | 客户名称 |
| 客户电话 | 文本 | 联系方式 |
| 产品需求 | 文本 | 需求描述 |
| 当前状态 | 单选 | 待处理 / 跟进中 / 已成交 / 已放弃 |
| 最后更新时间 | 日期时间 | 自动记录 |
| 确认人 | 文本 | 点击确认的销售姓名 |

### 3.2 获取表格 ID

打开多维表格，从 URL 复制：
```
https://xxx.feishu.cn/base/bascnXxxxx?table=tblXxxxx
                │                 │
              BASE_ID          TABLE_ID
```
