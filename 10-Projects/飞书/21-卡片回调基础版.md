# 飞书卡片回调基础版：点击同步多维表格

> 简化目标：用户点击按钮 → 选择状态 → 同步到多维表格

---

## 🎯 简化流程

```
┌─────────────┐      ┌─────────────┐      ┌─────────────┐
│  发送卡片    │ ───▶ │  用户点击    │ ───▶ │  更新表格    │
│  (埋记录ID) │      │  选择状态    │      │  同步状态    │
└─────────────┘      └─────────────┘      └─────────────┘
```

## 第二步：多维表格准备

### 表格字段

| 字段名 | 类型 | 说明 |
|--------|------|------|
| 客户姓名 | 文本 | - |
| 客户电话 | 文本 | - |
| 当前状态 | 单选 | 待处理 / 跟进中 / 已成交 / 已放弃 |
| 最后更新时间 | 日期时间 | 自动记录 |

### 获取表格信息

```bash
# 获取 base_id
curl https://open.feishu.cn/open-apis/bitable/v1/apps \
  -H "Authorization: Bearer {token}"
```

## 第四步：后端实现（简化版）

### 4.1 完整代码

```python
from flask import Flask, request, jsonify
import requests
import json
from datetime import datetime

app = Flask(__name__)

# 配置
APP_ID = "cli_xxxxxxxx"
APP_SECRET = "xxxxxxxx"
BASE_ID = "xxxxxxxx"      # 多维表格 ID
TABLE_ID = "xxxxxxxx"     # 表格 ID

# 缓存 token
tenant_token = None
token_expire = 0

def get_token():
    """获取 tenant_access_token"""
    global tenant_token, token_expire
    
    if tenant_token and datetime.now().timestamp() < token_expire:
        return tenant_token
    
    url = "https://open.feishu.cn/open-apis/auth/v3/tenant_access_token/internal/"
    resp = requests.post(url, json={"app_id": APP_ID, "app_secret": APP_SECRET})
    data = resp.json()
    
    tenant_token = data["tenant_access_token"]
    token_expire = datetime.now().timestamp() + data["expire"] - 300
    return tenant_token

def update_bitable(record_id, fields):
    """更新多维表格"""
    token = get_token()
    url = f"https://open.feishu.cn/open-apis/bitable/v1/apps/{BASE_ID}/tables/{TABLE_ID}/records/{record_id}"
    
    resp = requests.put(
        url,
        headers={"Authorization": f"Bearer {token}"},
        json={"fields": fields}
    )
    return resp.json()

# ============ 卡片回调处理 ============

@app.route('/webhook/feishu-card', methods=['POST'])
def handle_callback():
    data = request.json
    event = data.get('event', {})
    
    # 获取点击人信息
    operator = event.get('operator', {})
    clicker_name = operator.get('name')
    
    # 获取卡片数据
    value = event.get('action', {}).get('value', {})
    action = value.get('action')
    record_id = value.get('record_id')
    
    print(f"[点击] {clicker_name} 点击了 {action}, record_id={record_id}")
    
    if action == "update_status":
        # 返回选择状态的卡片
        return jsonify({
            "toast": {"type": "info", "content": "请选择新状态"},
            "card": {
                "schema": "2.0",
                "header": {
                    "template": "blue",
                    "title": {"tag": "plain_text", "content": "📝 选择新状态"}
                },
                "body": {
                    "direction": "vertical",
                    "elements": [
                        {"tag": "markdown", "content": f"当前操作人：**{clicker_name}**"},
                        {"tag": "hr"},
                        {
                            "tag": "column_set",
                            "columns": [
                                {
                                    "tag": "column",
                                    "width": "weighted",
                                    "weight": 1,
                                    "elements": [{
                                        "tag": "button",
                                        "text": {"tag": "plain_text", "content": "📞 跟进中"},
                                        "type": "primary",
                                        "behaviors": [{
                                            "type": "callback",
                                            "value": {"action": "set_status", "record_id": record_id, "status": "跟进中"}
                                        }]
                                    }]
                                },
                                {
                                    "tag": "column",
                                    "width": "weighted",
                                    "weight": 1,
                                    "elements": [{
                                        "tag": "button",
                                        "text": {"tag": "plain_text", "content": "✅ 已成交"},
                                        "type": "primary",
                                        "behaviors": [{
                                            "type": "callback",
                                            "value": {"action": "set_status", "record_id": record_id, "status": "已成交"}
                                        }]
                                    }]
                                },
                                {
                                    "tag": "column",
                                    "width": "weighted",
                                    "weight": 1,
                                    "elements": [{
                                        "tag": "button",
                                        "text": {"tag": "plain_text", "content": "❌ 已放弃"},
                                        "type": "danger",
                                        "behaviors": [{
                                            "type": "callback",
                                            "value": {"action": "set_status", "record_id": record_id, "status": "已放弃"}
                                        }]
                                    }]
                                }
                            ]
                        }
                    ]
                }
            }
        })
    
    if action == "set_status":
        # 更新多维表格
        status = value.get('status')
        
        result = update_bitable(record_id, {
            "当前状态": status,
            "最后更新时间": datetime.now().isoformat()
        })
        
        print(f"[更新] 状态={status}, 结果={result}")
        
        return jsonify({
            "toast": {"type": "success", "content": f"✅ 状态已更新为：{status}"},
            "card": {
                "schema": "2.0",
                "header": {
                    "template": "green",
                    "title": {"tag": "plain_text", "content": "✅ 状态更新成功"}
                },
                "body": {
                    "direction": "vertical",
                    "elements": [
                        {"tag": "markdown", "content": f"**操作人：**{clicker_name}\n**新状态：**{status}\n**时间：**{datetime.now().strftime('%m-%d %H:%M')}"}
                    ]
                }
            }
        })
    
    return jsonify({"toast": {"type": "info", "content": "已收到"}})

# ============ 发送卡片接口 ============

@app.route('/send-card', methods=['POST'])
def send_card():
    """发送卡片
    
    请求体：
    {
        "chat_id": "oc_xxxx",
        "customer_name": "张三",
        "phone": "13800138000",
        "record_id": "rec_xxxx"
    }
    """
    data = request.json
    token = get_token()
    
    card = {
        "schema": "2.0",
        "header": {
            "template": "blue",
            "title": {"tag": "plain_text", "content": f"📝 客户：{data['customer_name']}"}
        },
        "body": {
            "direction": "vertical",
            "elements": [
                {"tag": "markdown", "content": f"**电话：**{data['phone']}"},
                {
                    "tag": "button",
                    "text": {"tag": "plain_text", "content": "📝 更新状态"},
                    "type": "primary",
                    "behaviors": [{
                        "type": "callback",
                        "value": {"action": "update_status", "record_id": data['record_id']}
                    }]
                }
            ]
        }
    }
    
    resp = requests.post(
        "https://open.feishu.cn/open-apis/im/v1/messages",
        headers={"Authorization": f"Bearer {token}"},
        params={"receive_id_type": "chat_id"},
        json={
            "receive_id": data['chat_id'],
            "msg_type": "interactive",
            "content": json.dumps(card)
        }
    )
    return jsonify(resp.json())

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)
```

```
用户收到卡片
    ↓
点击「📝 更新状态」
    ↓
卡片变成三个选项：[📞 跟进中] [✅ 已成交] [❌ 已放弃]

用户选择一个状态
    ↓

    ↓
卡片变成绿色成功提示
```
